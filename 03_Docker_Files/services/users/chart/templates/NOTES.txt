PostgreSQL can be accessed via port 5432 on the following DNS name from within your cluster:

    {{ include "users.postgresql.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local

To get the password for postgres run:

    export POSTGRES_PASSWORD=$(kubectl get secret --namespace {{ .Release.Namespace }} {{ include "users.fullname" . }} -o jsonpath="{.data.postgres-password}" | base64 -d)

To connect to your database run the following command:

    kubectl run --namespace {{ .Release.Namespace }} {{ include "users.fullname" . }}-client --rm --tty -i --restart='Never' --image docker.io/bitnami/postgresql:16 --env="PGPASSWORD=$POSTGRES_PASSWORD" --command -- psql --host {{ include "users.postgresql.fullname" . }} -U {{ .Values.app.persistence.username }} -d {{ .Values.app.persistence.name }} -p 5432
{{- end }}
